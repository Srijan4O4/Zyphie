find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets)
add_library(zyph_desktop_ui STATIC window.cpp window.hpp)
target_link_libraries(zyph_desktop_ui PUBLIC zyphie::core Qt6::Widgets PRIVATE zyph_build_options)

  enable_language(RC)
  target_sources(zyphie PRIVATE version.rc)

if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(zyph_desktop_tests ../../tests/desktop/workflows.cpp)
  set_target_properties(zyph_desktop_tests PROPERTIES AUTOMOC ON)
  target_link_libraries(zyph_desktop_tests PRIVATE zyph_desktop_ui Qt6::Test)
  target_include_directories(zyph_desktop_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_definitions(zyph_desktop_tests PRIVATE ZYPH_CLI="$<TARGET_FILE:zyph>")
  add_test(NAME desktop_workflows COMMAND ${CMAKE_COMMAND} -DTEST_EXE=$<TARGET_FILE:zyph_desktop_tests> -P ${PROJECT_SOURCE_DIR}/tests/desktop/run.cmake)
  set_tests_properties(desktop_workflows PROPERTIES TIMEOUT 120)
endif()
