cmake_minimum_required(VERSION 3.25)
project(Zyphie VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(CTest)
option(ZYPH_ENABLE_SANITIZERS "Enable ASan and UBSan on GCC/Clang" OFF)
option(ZYPH_BUILD_DESKTOP "Build the Qt Widgets desktop application" OFF)

add_library(zyph_build_options INTERFACE)
if(MSVC)
  target_compile_options(zyph_build_options INTERFACE /W4 /permissive- /utf-8)
else()
  target_compile_options(zyph_build_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
endif()
if(ZYPH_ENABLE_SANITIZERS)
  if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Combined ASan/UBSan preset requires GCC or Clang with compatible sanitizer runtimes.")
  endif()
  target_compile_options(zyph_build_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(zyph_build_options INTERFACE -fsanitize=address,undefined)
endif()

find_package(Threads REQUIRED)
find_package(zstd CONFIG REQUIRED)
if(TARGET zstd::libzstd_static)
  set(ZYPH_ZSTD_TARGET zstd::libzstd_static)
elseif(TARGET zstd::libzstd_shared)
  set(ZYPH_ZSTD_TARGET zstd::libzstd_shared)
else()
  message(FATAL_ERROR "The installed zstd package exports no supported library target.")
endif()
add_library(zyph_core STATIC src/core/version.cpp src/core/digest.cpp src/format/format.cpp src/filesystem/io.cpp src/compression/codec.cpp src/archive/archive.cpp)
add_library(zyphie::core ALIAS zyph_core)
target_include_directories(zyph_core PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_compile_features(zyph_core PUBLIC cxx_std_20)
target_include_directories(zyph_core PRIVATE "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(zyph_core PRIVATE zyph_build_options ${ZYPH_ZSTD_TARGET} Threads::Threads)
if(WIN32)
  target_link_libraries(zyph_core PRIVATE bcrypt advapi32 ntdll)
else()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(zyph_core PRIVATE OpenSSL::Crypto)
endif()

add_executable(zyph apps/cli/main.cpp)
target_link_libraries(zyph PRIVATE zyphie::core zyph_build_options)
if(MINGW)
  target_link_options(zyph PRIVATE -municode -static)
endif()

if(BUILD_TESTING)
  target_compile_definitions(zyph_core PRIVATE ZYPH_TEST_HOOKS)

  find_package(GTest CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(zyph_foundation_tests tests/unit/foundation_tests.cpp tests/format/format_tests.cpp tests/roundtrip/archive_tests.cpp)
  target_link_libraries(zyph_foundation_tests PRIVATE zyphie::core zyph_build_options
    GTest::gtest_main ${ZYPH_ZSTD_TARGET} Threads::Threads)
  target_compile_definitions(zyph_foundation_tests PRIVATE ZYPH_FIXTURES="${PROJECT_SOURCE_DIR}/tests/fixtures")
  target_compile_definitions(zyph_foundation_tests PRIVATE ZYPH_TEST_HOOKS)
  target_include_directories(zyph_foundation_tests PRIVATE src)
  include(GoogleTest)
  gtest_discover_tests(zyph_foundation_tests DISCOVERY_TIMEOUT 30)
  add_test(NAME cli_roundtrip COMMAND ${CMAKE_COMMAND} -D ZYPH=$<TARGET_FILE:zyph> -D TEST_ROOT=${PROJECT_BINARY_DIR}/integration -P ${PROJECT_SOURCE_DIR}/tests/integration/cli.cmake)
  add_test(NAME cli_version COMMAND zyph --version)
  set_tests_properties(cli_version PROPERTIES PASS_REGULAR_EXPRESSION "1.0.0")
  add_test(NAME cli_help COMMAND zyph --help)
  set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "Overwrite is not supported")
  add_test(NAME cli_rejects_missing_pack_arguments COMMAND zyph pack)
  set_tests_properties(cli_rejects_missing_pack_arguments PROPERTIES WILL_FAIL TRUE)
endif()

option(ZYPH_BUILD_FUZZERS "Build Linux Clang libFuzzer targets" OFF)
if(ZYPH_BUILD_DESKTOP)
  add_subdirectory(apps/desktop)
endif()
if(ZYPH_BUILD_FUZZERS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT ZYPH_ENABLE_SANITIZERS)
    message(FATAL_ERROR "Fuzzing requires Linux Clang and ZYPH_ENABLE_SANITIZERS")
  endif()
  target_compile_options(zyph_core PRIVATE -fsanitize=fuzzer-no-link)
  foreach(kind archive path)
    add_executable(zyph_fuzz_${kind} tests/fuzz/${kind}_fuzz.cpp)
    target_link_libraries(zyph_fuzz_${kind} PRIVATE zyphie::core zyph_build_options)
    target_compile_options(zyph_fuzz_${kind} PRIVATE -fsanitize=fuzzer)
    target_link_options(zyph_fuzz_${kind} PRIVATE -fsanitize=fuzzer)
  endforeach()
endif()
